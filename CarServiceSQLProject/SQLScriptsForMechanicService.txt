CREATE TABLE CUSTOMER (
    CusID        VARCHAR(10),
    Forename     VARCHAR(50),
    Surname      VARCHAR(50),
    Email        VARCHAR(100),
    PhoneNo      VARCHAR(20),
    CONSTRAINT PK_Customer PRIMARY KEY (CusID)
);

CREATE TABLE CAR (
    RegNum       VARCHAR(15),
    Make         VARCHAR(50),
    Model        VARCHAR(50),
    Manufacture  YEAR,
    CusID        VARCHAR(10),
    CONSTRAINT PK_Car PRIMARY KEY (RegNum),
    CONSTRAINT FK_Car_Customer FOREIGN KEY (CusID)
        REFERENCES CUSTOMER(CusID)
);

CREATE TABLE MECHANIC (
    MechanicID   VARCHAR(10),
    Forename     VARCHAR(50),
    Surname      VARCHAR(50),
    PhoneNo      VARCHAR(20),
    Grade        VARCHAR(20),
    CONSTRAINT PK_Mechanic PRIMARY KEY (MechanicID)
);

CREATE TABLE SERVICE (
    ServiceID        VARCHAR(10),
    DropOffDate      DATE,
    DropOffTime      TIME,
    WorkDescription  VARCHAR(500),
    Mileage          INT,
    DateOfNextService DATE,
    RegNum           VARCHAR(15),
    CONSTRAINT PK_Service PRIMARY KEY (ServiceID),
    CONSTRAINT FK_Service_Car FOREIGN KEY (RegNum)
        REFERENCES CAR(RegNum)
);

CREATE TABLE PERFORMS (
    ServiceID   VARCHAR(10),
    MechanicID  VARCHAR(10),
    TimeOfService    Time,
    CONSTRAINT PK_Performs PRIMARY KEY (ServiceID, MechanicID),
    CONSTRAINT FK_Performs_Service FOREIGN KEY (ServiceID)
        REFERENCES SERVICE(ServiceID),
    CONSTRAINT FK_Performs_Mechanic FOREIGN KEY (MechanicID)
        REFERENCES MECHANIC(MechanicID)
);
